# The credential fixture uses exact current UI method bodies in an isolated
# shell. Generation is confined to the build tree and records source/body hashes.
set(credential_method_source "${CMAKE_CURRENT_SOURCE_DIR}/../../desktop/src/MainWindow.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${credential_method_source}")
file(READ "${credential_method_source}" credential_source)
string(FIND "${credential_source}" "QWidget* MainWindow::credentialCopyActions(" credential_begin)
string(FIND "${credential_source}" "void MainWindow::removeAccount(" credential_end)
if(credential_begin LESS 0 OR credential_end LESS credential_begin)
  message(FATAL_ERROR "Credential method boundaries were not found")
endif()
math(EXPR credential_length "${credential_end} - ${credential_begin}")
string(SUBSTRING "${credential_source}" ${credential_begin} ${credential_length} credential_methods)
file(SHA256 "${credential_method_source}" credential_source_sha256)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/credential-review-methods.inc" "${credential_methods}")
file(SHA256 "${CMAKE_CURRENT_BINARY_DIR}/credential-review-methods.inc" credential_methods_sha256)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/credential-method-binding.json"
  "{\"source_sha256\":\"${credential_source_sha256}\",\"methods_sha256\":\"${credential_methods_sha256}\"}\n")
add_executable(credential_deadline credential_deadline.cpp ../../desktop/src/ApiClient.hpp)
target_include_directories(credential_deadline PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(credential_deadline PRIVATE Qt6::Core Qt6::Network Qt6::Widgets)
add_test(NAME credential_deadline COMMAND credential_deadline)
set_tests_properties(credential_deadline PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 10)

# Exact exit/restart/library/update methods; no real updater or service is started.
set(campaign_exit_methods "${campaign_close_methods}")
foreach(exit_pair IN ITEMS "installUpdate|startBackend" "restartBackend|probeBackend" "chooseLibrary|exportDiagnostics")
  string(REPLACE "|" ";" exit_boundaries "${exit_pair}")
  list(GET exit_boundaries 0 exit_first)
  list(GET exit_boundaries 1 exit_last)
  string(FIND "${campaign_preview_source}" "void MainWindow::${exit_first}(" exit_begin)
  string(FIND "${campaign_preview_source}" "void MainWindow::${exit_last}(" exit_end)
  if(exit_begin LESS 0 OR exit_end LESS exit_begin)
    message(FATAL_ERROR "Campaign exit method boundaries were not found: ${exit_pair}")
  endif()
  math(EXPR exit_length "${exit_end} - ${exit_begin}")
  string(SUBSTRING "${campaign_preview_source}" ${exit_begin} ${exit_length} exit_method)
  string(APPEND campaign_exit_methods "${exit_method}")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/campaign-exit-methods.inc" "${campaign_exit_methods}")
file(SHA256 "${CMAKE_CURRENT_BINARY_DIR}/campaign-exit-methods.inc" campaign_exit_body_sha256)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/campaign-exit-method-binding.json"
  "{\"source_sha256\":\"${credential_source_sha256}\",\"methods_sha256\":\"${campaign_exit_body_sha256}\"}\n")
add_executable(campaign_exit_paths campaign_exit_paths.cpp)
target_include_directories(campaign_exit_paths PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(campaign_exit_paths PRIVATE Qt6::Core Qt6::Network Qt6::Widgets)
foreach(exit_mode IN ITEMS restart restart-close update update-failure update-close-failure quit library-reentrant)
  add_test(NAME campaign_exit_${exit_mode} COMMAND campaign_exit_paths ${exit_mode})
  set_tests_properties(campaign_exit_${exit_mode} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 10)
endforeach()
